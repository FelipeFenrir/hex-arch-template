package com.acme.personmdm.domain.common.specification;

import com.acme.personmdm.domain.common.validation.FailNotification;

import java.util.ArrayList;
import java.util.List;

public class SpecValidator<T> {
    List<GenericSpecification<T>> specs = new ArrayList<>();

    public void addSpec(GenericSpecification<T> spec) {
        specs.add(spec);
    }

    public SpecValidator<T> addSpecs(List<GenericSpecification<T>> specs) {
        for (GenericSpecification<T> spec : specs) {
            addSpec(spec);
        }
        return this;
    }

    public FailNotification validate(T target) {
        FailNotification notification = new FailNotification();

        specs.stream()
                .filter(spec -> !spec.isSatisfiedBy(target))
                .map(GenericSpecification::message)
                .forEach(notification::addError);

        return notification;
    }
}
