package com.acme.personmdm.domain.common.specification;

import java.util.function.Predicate;

public record GenericSpecification<T>(Predicate<T> predicate, String message) implements Specification<T> {

    @Override
    public boolean isSatisfiedBy(T candidate) {
        return predicate.test(candidate);
    }
}
