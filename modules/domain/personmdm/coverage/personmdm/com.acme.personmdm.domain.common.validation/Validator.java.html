package com.acme.personmdm.domain.common.validation;

import java.util.ArrayList;
import java.util.List;

public class Validator<T> {
    private final List<Rule<T>> rules = new ArrayList<>();

    public Validator<T> addRule(Rule<T> rule) {
        rules.add(rule);
        return this;
    }

    public Validator<T> addRules(Rule<T>... rules) {
        for (Rule<T> rule : rules) {
            addRule(rule);
        }
        return this;
    }

    public Validator<T> addRules(List<Rule<T>> rules) {
        for (Rule<T> rule : rules) {
            addRule(rule);
        }
        return this;
    }

    public FailNotification validate(T target) {
        FailNotification notification = new FailNotification();
        for (Rule<T> rule : rules) {
            if (rule.test(target)) {
                notification.addError(rule.message());
            }
        }
        return notification;
    }
}
